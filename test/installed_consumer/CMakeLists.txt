cmake_minimum_required(VERSION 3.19)
project(MotorControlCANConsumer LANGUAGES CXX)
if(MOTORCONTROLCAN_SOURCE_DIR)
    # Parent settings must survive add_subdirectory without package tests.
    set(BUILD_TESTING OFF)
    add_subdirectory("${MOTORCONTROLCAN_SOURCE_DIR}" core)
    if(BUILD_TESTING)
        message(FATAL_ERROR "Library changed parent BUILD_TESTING")
    endif()
else()
    if(MOTORCONTROLCAN_EXPECTED_VERSION)
        find_package(MotorControlCAN ${MOTORCONTROLCAN_EXPECTED_VERSION} EXACT CONFIG REQUIRED)
    else()
        find_package(MotorControlCAN CONFIG REQUIRED)
    endif()
endif()
get_target_property(interface_options MotorControlCAN::MotorControlCAN INTERFACE_COMPILE_OPTIONS)
if(interface_options)
    message(FATAL_ERROR "Private compiler options leaked to consumer: ${interface_options}")
endif()

function(strict_consumer target standard)
    target_link_libraries(${target} PRIVATE MotorControlCAN::MotorControlCAN)
    set_target_properties(${target} PROPERTIES
        CXX_STANDARD ${standard} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
        if(standard EQUAL 17)
            target_compile_options(${target} PRIVATE /GR-)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -pedantic-errors)
        if(standard EQUAL 17)
            target_compile_options(${target} PRIVATE -fno-rtti)
        endif()
    endif()
endfunction()
foreach(standard 11 17)
    add_executable(consumer_${standard} main.cpp)
    strict_consumer(consumer_${standard} ${standard})
    file(GLOB_RECURSE headers RELATIVE "${MOTORCONTROLCAN_HEADERS_DIR}"
        "${MOTORCONTROLCAN_HEADERS_DIR}/MotorControlCAN/*.h")
    if(NOT headers)
        message(FATAL_ERROR "No package headers to check")
    endif()
    set(header_sources)
    foreach(header IN LISTS headers)
        string(MAKE_C_IDENTIFIER "${header}" header_id)
        set(source "${CMAKE_CURRENT_BINARY_DIR}/headers_${standard}/${header_id}.cpp")
        file(WRITE "${source}" "#include <${header}>\n")
        list(APPEND header_sources "${source}")
    endforeach()
    add_library(headers_${standard} OBJECT ${header_sources})
    strict_consumer(headers_${standard} ${standard})
endforeach()
enable_testing()
add_test(NAME consumer_11 COMMAND consumer_11)
add_test(NAME consumer_17_no_rtti COMMAND consumer_17)
