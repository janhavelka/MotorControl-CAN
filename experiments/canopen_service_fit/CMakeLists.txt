cmake_minimum_required(VERSION 3.19)
project(canopen_service_fit LANGUAGES C CXX)

if(NOT TARGET canopennode_sdo)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/UPSTREAM.json" upstream_manifest)
string(JSON upstream_revision GET "${upstream_manifest}" revision)
if(NOT upstream_revision STREQUAL "9b8beed8367241e96ac03f916cd5a500bcb2cf23")
    message(FATAL_ERROR "Unexpected CANopenNode revision in UPSTREAM.json")
endif()
string(JSON upstream_count LENGTH "${upstream_manifest}" files)
math(EXPR upstream_last "${upstream_count} - 1")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/UPSTREAM.json")
foreach(index RANGE 0 ${upstream_last})
    string(JSON relative_path GET "${upstream_manifest}" files ${index} path)
    string(JSON expected_hash GET "${upstream_manifest}" files ${index} sha256)
    set(input_path "${CMAKE_CURRENT_SOURCE_DIR}/${relative_path}")
    file(SHA256 "${input_path}" actual_hash)
    if(NOT actual_hash STREQUAL expected_hash)
        message(FATAL_ERROR "CANopenNode SHA-256 mismatch: ${relative_path}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input_path}")
endforeach()

enable_testing()
set(upstream "${CMAKE_CURRENT_SOURCE_DIR}/vendor/CANopenNode")
add_library(canopennode_sdo STATIC

    "${upstream}/301/CO_NMT_Heartbeat.c"

    "${upstream}/301/CO_fifo.c")
target_include_directories(canopennode_sdo PUBLIC "${upstream}" "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(canopennode_sdo PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endif()
