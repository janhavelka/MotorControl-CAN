cmake_minimum_required(VERSION 3.19)
if(NOT TARGET MotorControlCANProbeApp)
    message(FATAL_ERROR "Build action_preview from the repository root with examples and CANopenNode enabled")
endif()
add_executable(action_preview main.cpp)
target_link_libraries(action_preview PRIVATE MotorControlCANProbeApp)
set_target_properties(action_preview PROPERTIES
    CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(action_preview PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(action_preview PRIVATE -Wall -Wextra -Wpedantic -Werror -pedantic-errors)
endif()
