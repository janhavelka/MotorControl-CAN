cmake_minimum_required(VERSION 3.19)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MotorControlCANDriverSettingsPreview LANGUAGES CXX)
endif()
if(NOT TARGET MotorControlCAN::MotorControlCAN)
    find_package(MotorControlCAN CONFIG REQUIRED)
endif()
add_executable(driver_settings_preview main.cpp)
target_link_libraries(driver_settings_preview PRIVATE MotorControlCAN::MotorControlCAN)
set_target_properties(driver_settings_preview PROPERTIES
    CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(driver_settings_preview PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(driver_settings_preview PRIVATE -Wall -Wextra -Wpedantic -Werror -pedantic-errors)
endif()
