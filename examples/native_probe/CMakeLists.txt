cmake_minimum_required(VERSION 3.19)
if(NOT TARGET MotorControlCANProbeApp)
    message(FATAL_ERROR "Build native_probe from the root with examples and CANopenNode enabled")
endif()
add_library(MotorControlCANFakeProbe STATIC FakeTransport.cpp)
target_include_directories(MotorControlCANFakeProbe PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(MotorControlCANFakeProbe PUBLIC MotorControlCANProbeApp)
add_executable(native_probe main.cpp)
target_link_libraries(native_probe PRIVATE MotorControlCANFakeProbe)
foreach(target MotorControlCANFakeProbe native_probe)
    set_target_properties(${target} PROPERTIES
        CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
