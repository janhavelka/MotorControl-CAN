cmake_minimum_required(VERSION 3.19)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MotorControlCANPositionPreview LANGUAGES CXX)
endif()
if(NOT TARGET MotorControlCAN::MotorControlCAN)
    find_package(MotorControlCAN CONFIG REQUIRED)
endif()
add_executable(position_preview main.cpp)
target_link_libraries(position_preview PRIVATE MotorControlCAN::MotorControlCAN)
set_target_properties(position_preview PROPERTIES
    CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(position_preview PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(position_preview PRIVATE -Wall -Wextra -Wpedantic -Werror -pedantic-errors)
endif()
