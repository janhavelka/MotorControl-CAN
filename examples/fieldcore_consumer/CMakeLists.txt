cmake_minimum_required(VERSION 3.19)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MotorControlCANFieldCoreConsumer LANGUAGES CXX)
endif()
if(NOT TARGET MotorControlCAN::MotorControlCAN)
    find_package(MotorControlCAN CONFIG REQUIRED)
endif()
add_library(MotorControlCANFieldCoreConsumer STATIC MotorModule.cpp)
target_include_directories(MotorControlCANFieldCoreConsumer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(MotorControlCANFieldCoreConsumer PUBLIC MotorControlCAN::MotorControlCAN)
add_executable(fieldcore_consumer main.cpp)
target_link_libraries(fieldcore_consumer PRIVATE MotorControlCANFieldCoreConsumer)
foreach(target MotorControlCANFieldCoreConsumer fieldcore_consumer)
    set_target_properties(${target} PROPERTIES
        CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -pedantic-errors)
    endif()
endforeach()
